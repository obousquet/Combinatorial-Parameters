\subsection{Basic}\label{sec:Basic}
\subsubsection{Effective Range}
      
      \begin{definition}[Effective Range - {$\mathrm{E}$}]\label{def:effective_range}
      The {\em effective range} $E(\H)$ of a family $\H$ is the number of elements that can be distinguished by members of $\H$: \[ E(\H)\bydef \abs*{\set*{x:x\in\X,\, \exists h_1,h_2\in\H,\, x\in h_1 \Delta h_2}} \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: SM
      \end{itemize}
\subsubsection{Size}
      
      \begin{definition}[Size - {$\mathrm{H}$}]\label{def:size}
      The {\em size} of a family $\H$ is simply the  cardinality, \[  \abs*{\H} \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: SM
      \end{itemize}
\subsubsection{Oriented Diameter}
      
      \begin{definition}[Oriented Diameter - {$\mathrm{D}^+$}]\label{def:oriented_diameter}
      The {\em oriented diameter} $\mathrm{D}^+(\H)$ of a family $\H$ is defined as \[ \mathrm{D}^+(\H)\bydef \max\set*{\abs*{S}: \set*{\emptyset, S}\subset \H_{|S}} \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: N
      \item {\bf Monotonicity Type}: DM
      \end{itemize}
\subsubsection{Diameter}
      
      \begin{definition}[Diameter - {$\mathrm{D}$}]\label{def:diameter}
      The {\em diameter} $D(\H)$ of a family $\H$ is the largest distance between two elements of the family, i.e. \[ D(\H)\bydef \max_{h_1,h_2\in\H} \abs*{h_1\Delta h_2} \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \item {\bf Notes}: The diameter is equal to the strict diameter dimension.
\end{itemize}
\subsubsection{Hamming Radius}
      
      \begin{definition}[Hamming Radius - {$\mathrm{B}$}]\label{def:hamming_radius}
      The {\em Hamming Radius} $B(\H)$ of a family $\H$ is the smallest radius of an enclosing Hamming ball, defined as \[ B(\H)\bydef \min\set*{d: \exists T\subset \X,\, \forall h\in \H,\, |h\Delta T|\le d} \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \end{itemize}
\subsubsection{Threshold Dimension}
      
      \begin{definition}[Threshold Dimension - {$\mathrm{T}$}]\label{def:threshold_dimension}
      The {\em Threshold dimension} of a family $\H$, denoted by $T(\H)$, is the largest integer $d$ such that there exists a set $S\subset \X$, of size $d$ such that $\H_{|S}$ contains thresholds, i.e. there is an ordering $x_1,\ldots,x_d$ of the elements of $S$ such that \[ \forall 0\le k\le d,\, \set*{x_1,\ldots,x_k} \in \H_{|S}\,. \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: N
      \item {\bf Monotonicity Type}: DM
      \item {\bf Notes}: Note that this means $\H_{|S}$ contains a $d+1$-chain, i.e. $d+1$ sets $S_1,\ldots,S_{d+1}$ such that $S_1\subset \ldots\subset S_{d+1}$.
\end{itemize}
\subsubsection{Path Dimension}
      
      \begin{definition}[Path Dimension - {$\mathrm{P}$}]\label{def:path_dimension}
      The {\em Path dimension} of a family $\H$, denoted by $P(\H)$, is the largest integer $d$ such that there exists a set $S\subset \X$, of size $d$ such that $\H_{|S}$ contains a full path, i.e. there is an ordering $x_1,\ldots,x_d$ of the elements of $S$ and a subset $T\subset S$ such that \[ \forall 0\le k\le d,\, T\Delta \set*{x_1,\ldots,x_k} \in \H_{|S}\,. \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \end{itemize}

\subsubsection{Information Complexity}
      
      \begin{definition}[Information Complexity - {$\mathrm{IC}$}]\label{def:information_complexity}
      For a finite Boolean class $\mathcal H$, its \emph{information complexity} is
      \[
      \mathrm{IC}(\mathcal H)
      =\sup_{m\geq1}\inf_{\mathcal A}\sup_{D}
      I(S;\mathcal A(S)),
      \]
      where the infimum is over (possibly randomized, improper) algorithms
      that PAC learn $\mathcal H$, $D$ ranges over realizable distributions,
      and $S\sim D^m$.  It is monotone under both subfamilies and coordinate
      projections: a learner for $\mathcal H$ can be used on a projected
      sample and its output restricted to the retained coordinates.  If
      $\operatorname{Ldim}(\mathcal H)=d$, then Pradeep, Nachum, and Gastpar
      prove the quantitative bound
      \[
      \mathrm{IC}(\mathcal H)\leq
      2^d+\log_2(d+1)+3+\frac{3}{e\ln 2}.
      \]
      In their model, the qualitative converse also holds for arbitrary
      binary classes: information complexity is finite exactly when
      Littlestone dimension is finite.
      This is exponential in $d$, and hence is recorded as prose rather than
      as one of the catalogue's affine or logarithmic relationship types.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: nM
      \item {\bf References}: https://arxiv.org/abs/2206.13257
\end{itemize}

      \begin{definition}[VC Dimension - {$\mathrm{VC}$}]\label{def:vc_dimension}
      The {\em VC dimension} of a family $\H$, denoted by $\VC(\H)$, is the largest integer $d$ such that there exists a set $S\subset \X$, of size $d$ such that $S$ is shattered by $\H$, i.e.  \[ \abs*{\H_{|S}} = 2^{\abs*{S}}\,. \]

      For $\H\subseteq 2^\X$, let $\H^*=\set*{x^*:x\in\X}$ be the dual class on domain $\H$, where $x^*=\set*{h\in\H:x\in h}$.  The \emph{dual VC dimension} is $\mathrm{VC}^{*}(\H)=\mathrm{VC}(\H^*)$.
      \end{definition}

      \item {\bf Bit-flipping invariant}: Y

\subsubsection{Dual Antipodal VC Dimension}
      \begin{definition}[Dual Antipodal VC Dimension - {$\mathrm{VC}^{*a}$}]\label{def:dual_antipodal_vc_dimension}
      A set of hypotheses $A\subseteq\H$ is dually antipodally shattered if, for every labeling $s:A\to\{0,1\}$, some point induces either $s$ or its bitwise complement on $A$.  The \emph{dual antipodal VC dimension} $\mathrm{VC}^{*a}(\H)$ is the largest size of such a set.

\subsubsection{Spherical Dimension}
      \begin{definition}[Spherical Dimension - {$\mathrm{sd}$}]\label{def:spherical_dimension}
      Let $\H$ be a finite binary class.  Write $\Delta_\H$ for its space of realizable distributions on labeled examples, and $\Delta_\H^{\mathrm{ant}}=\Delta_\H\cap(-\Delta_\H)$, where $-$ flips labels.  The \emph{spherical dimension} $\mathrm{sd}(\H)$ is the largest $d$ for which an antipodality-preserving continuous map $S^d\to\Delta_\H^{\mathrm{ant}}$ exists.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y

\subsubsection{Combinatorial Eluder Dimension}
      \begin{definition}[Combinatorial Eluder Dimension - {$\mathrm{Edim}$}]\label{def:combinatorial_eluder_dimension}
      For a binary class $\H$ and $h^*\in\H$, let $\mathrm{Edim}_{h^*}(\H)$ be the largest $m$ for which there are points $x_1,\ldots,x_m$ and hypotheses $h_1,\ldots,h_m\in\H$ satisfying $h_i(x_i)\ne h^*(x_i)$ and $h_i(x_j)=h^*(x_j)$ for every $j<i$. \subsection{Algebraic}\label{sec:Algebraic}
\subsubsection{Yang dimension}
      Default characteristic-two Yang dimension
      \begin{definition}[Yang dimension - {$\mathrm{Y}$}]\label{def:yang_dimension}
      Let $\Delta_{\mathcal H}$ be the simplicial complex of realizable partial labelings of a finite Boolean class $\mathcal H$.  In this catalogue, the \emph{Yang dimension} is computed over $\mathbb F_2$:
      \[
      \mathrm Y(\mathcal H)=\mathrm Y_2(\mathcal H)
      :=\operatorname{pd}_{\mathbb F_2}(I^*_{\mathcal H})
      =\operatorname{reg}_{\mathbb F_2}\mathbb F_2[\Delta_{\mathcal H}].
      \]
      Thus it is the length of a minimal free resolution of Yang's canonical ideal over $\mathbb F_2$.  It satisfies $\mathrm Y_0(\mathcal H)\leq\mathrm Y(\mathcal H)$.  The homological formula permits exceptions at torsion primes, but no explicit Boolean class separating $\mathrm Y_2$ from another characteristic is currently recorded.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: pM
      \end{itemize}
\subsubsection{Characteristic-p Yang Dimension}
      \begin{definition}[Characteristic-p Yang Dimension - {$\mathrm{Y}_p$}]\label{def:characteristic_p_yang_dimension}
      For each prime $p$, define
      \[
      \mathrm Y_p(\mathcal H):=\operatorname{pd}_{\mathbb F_p}(I^*_{\mathcal H})
      =\operatorname{reg}_{\mathbb F_p}\mathbb F_p[\Delta_{\mathcal H}].
      \]
      Extension of scalars shows that the value depends only on the characteristic, not on the chosen field of characteristic $p$.  Universal coefficients give
      \[
      \mathrm Y_p(\mathcal H)\geq\mathrm Y_0(\mathcal H)
      \]
      for every prime $p$.  For a fixed finite class, equality holds outside a finite set of primes: the possible exceptional primes divide torsion in an integral homology group of one of the Hochster/interval complexes.  Thus the homological formula permits characteristic dependence.  No explicit Boolean class with $\mathrm Y_p\ne\mathrm Y_q$ is currently recorded, so neither a strict separation nor an ordering between distinct positive characteristics is asserted.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: pM
      \end{itemize}
\subsubsection{Characteristic-zero Yang Dimension}
      \begin{definition}[Characteristic-zero Yang Dimension - {$\mathrm{Y}_0$}]\label{def:characteristic_zero_yang_dimension}
      Define
      \[
      \mathrm Y_0(\mathcal H):=\operatorname{pd}_{\mathbb Q}(I^*_{\mathcal H})
      =\operatorname{reg}_{\mathbb Q}\mathbb Q[\Delta_{\mathcal H}].
      \]
      This has the same value over every field of characteristic zero.  For every fixed finite class, it agrees with $\mathrm Y_p$ for all but finitely many primes $p$.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: pM
      \end{itemize}

\subsubsection{Stable Unlabeled Complexity}
      
      \begin{definition}[Stable Unlabeled Complexity - {$\mathrm{sUSC}$}]\label{def:stable_unlabeled_complexity}
      An unlabeled sample-compression scheme $(\kappa,\rho)$ is \emph{stable} if, for every realizable sample $S$ and every subsample $S'$ satisfying $\kappa(S)\subseteq S'\subseteq S$, its reconstructed hypothesis is unchanged:
      \[
          \rho(\kappa(S'))=\rho(\kappa(S)).

\subsubsection{Littlestone Dimension}
      
      \begin{definition}[Littlestone Dimension - {$\mathrm{L}$}]\label{def:littlestone_dimension}
      A \emph{complete binary decision tree} has a domain point at every internal node and edges labeled $0$ and $1$.  It is \emph{shattered} by $\H$ if every root-to-leaf sequence of queried points and edge labels is realized by a member of $\H$.  The \emph{Littlestone dimension} $\mathrm L(\H)$ is the supremum of the common root-to-leaf depth of such complete shattered trees.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: SM
      \end{itemize}
\subsubsection{Randomized Littlestone Dimension}
      
      \begin{definition}[Randomized Littlestone Dimension - {$\mathrm{RL}$}]\label{def:randomized_littlestone_dimension}
      Let $T$ be a finite binary decision tree shattered by $\H$.  Starting at its root, choose independently and uniformly one of the two outgoing edges at each internal node, and let $E_T$ be the resulting root-to-leaf length.  The \emph{randomized Littlestone dimension} is
      \[
        \mathrm{RL}(\H) \bydef \frac12\sup_{T\text{ shattered by }\H} E_T.
      \]
      Thus it replaces the minimum branch length in the ordinary Littlestone definition by the average branch length, with the normalization that equals the optimal randomized realizable mistake bound.  This is the notion of Filmus, Hanneke, Mehalel, and Moran \cite{filmus2023randomized}.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: SM
      \end{itemize}

\subsubsection{$k$-Mistake Littlestone Dimension}
      
      \begin{definition}[$k$-Mistake Littlestone Dimension - {$\mathrm{L}^{\mathrm{mist}}_k$}]\label{def:kmistake_littlestone_dimension}
      For $k\ge0$, a labeled sequence is \emph{$k$-realizable} by $\H$ if some $h\in\H$ disagrees with it on at most $k$ queried labels.  A binary decision tree is $k$-shattered by $\H$ when each root-to-leaf branch is $k$-realizable.  The \emph{$k$-mistake Littlestone dimension} is the supremum, over all such trees, of their minimum root-to-leaf branch length.  Thus it is not a complete-tree depth in general.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \end{itemize}
\subsubsection{Randomized $k$-Mistake Littlestone Dimension}
      
      \begin{definition}[Randomized $k$-Mistake Littlestone Dimension - {$\mathrm{RL}^{\mathrm{mist}}_k$}]\label{def:randomized_kmistake_littlestone_dimension}
      Fix $k\ge0$.  A binary decision tree is \emph{$k$-shattered} by $\H$ if every root-to-leaf labeled sequence disagrees with some $h\in\H$ on at most $k$ queried labels.  Let $E_T$ be the root-to-leaf length when a branch of $T$ is followed by independent uniform choices at every internal node.  The \emph{randomized $k$-mistake Littlestone dimension} is
      \[
        \mathrm{RL}^{\mathrm{mist}}_k(\H)\bydef\frac12\sup_{T\text{ $k$-shattered by }\H}E_T.
      \]
      It is the optimal expected mistake bound in the $k$-realizable online-learning setting of Filmus, Hanneke, Mehalel, and Moran \cite{filmus2023randomized}.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \end{itemize}
\subsubsection{Prefix-$k$ Littlestone Dimension}
      
      \begin{definition}[Prefix-$k$ Littlestone Dimension - {$\mathrm{PL}_k$}]\label{def:prefix_littlestone_dimension}
      Fix $k\ge1$.  A \emph{prefix-$k$ Littlestone tree} is an ordinarily shattered complete binary tree in which the coordinate labels are horizontally constant on each of its first $\min\{k,d\}$ levels, where $d$ is its depth; all subsequent labels are unrestricted.  Its maximum possible depth is the \emph{prefix-$k$ Littlestone dimension} $\mathrm{PL}_k(\H)$.  In particular, $\mathrm{PL}_1(\H)=\mathrm L(\H)$.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \end{itemize}
\subsection{Mixed}\label{sec:Mixed}
\subsubsection{Star number = Maximum Projected Teaching Set Size = Projected Maximal Degree}
      
      \begin{definition}[Star number = Maximum Projected Teaching Set Size = Projected Maximal Degree - {\begin{tabular}{l}$\mathrm{S} = \mathrm{TS}_{\max}^p = \mathrm{d}^p_{\max}$\end{tabular}}]\label{def:star_number_=_maximum_projected_teaching_set_size_=_projected_maximal_degree}
      The {\em Star dimension} of a family $\H$, denoted by $S(\H)$, is the largest integer $d$ such that there exists a set $S\subset \X$, of size $d$ such that $\H_{|S}$ contains a $d$-star, i.e. there is a subset $T\subset S$ such that \[ T\in \H_{|S} \,\mbox{ and }\, \forall x\in S,\, T\Delta \set*{x} \in \H_{|S}\,. \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \end{itemize}
\subsubsection{Largest Shattered Set = VC Dimension = Projected VC Radius = Projected Strongly Shattered Dimension = Maximum Largest Order Shattered Set}
      
      \begin{definition}[Largest Shattered Set = VC Dimension = Projected VC Radius = Projected Strongly Shattered Dimension = Maximum Largest Order Shattered Set - {\begin{tabular}{l}$\mathrm{SH} = \mathrm{VC} = \mathrm{VCR}^p$\\$= \mathrm{ST}^p = \mathrm{OSH}_{\max}$\end{tabular}}]\label{def:largest_shattered_set_=_vc_dimension_=_projected_vc_radius_=_projected_strongly_shattered_dimension_=_maximum_largest_order_shattered_set}
      The \emph{largest shattered set} is the maximum cardinality of a subset of the domain shattered by $\H$; equivalently, it is the VC dimension of $\H$.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: DM
      \end{itemize}
\subsubsection{Minimum Star Number}

      \begin{definition}[Minimum Star Number - {$\mathrm{S}_{\mathrm{min}}$}]\label{def:minimum_star_number}
      For a possibly unrealizable labeling $g:\X\to\{0,1\}$, let $S_g(\H)$ be the largest $d$ for which there are distinct points $x_1,\ldots,x_d$ and $h_0,\ldots,h_d\in\H$ such that $h_0(x_j)=g(x_j)$ for all $j$, while $h_i(x_j)=g(x_j)$ holds exactly when $j\ne i$ for each $i\ge1$.  The \emph{minimum star number} is
      \[
      \mathrm{S}_{\mathrm{min}}(\H)\bydef\inf_{g:\X\to\{0,1\}}S_g(\H).
      \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: M
      \item {\bf References}: \cite{hanneke2024star_eluder}
      \end{itemize}
\subsubsection{Radon Number}

      \begin{definition}[Radon Number - {$\mathrm{Rad}$}]\label{def:radon_number}
      Associate to a finite binary class $\H$ the convexity space on ground set $\H$ whose convex sets are the version spaces $\{h\in\H:h|_S=y\}$ and their intersections.  A subset $\{h_1,\ldots,h_k\}\subseteq\H$ is \emph{Radon independent} if, for every nontrivial partition $I\sqcup J=[k]$, the convex hulls of $\{h_i:i\in I\}$ and $\{h_j:j\in J\}$ are disjoint.  The \emph{Radon number} $\mathrm{Rad}(\H)$ is the largest size of a Radon-independent subset.
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf References}: \cite{chase2024dual}
      \end{itemize}
\subsection{Holes and Homology}\label{sec:Holes and Homology}
\subsubsection{co-VC dimension}
      
      \begin{definition}[co-VC dimension - {$\mathrm{coVC}$}]\label{def:co-vc_dimension}
      The {\em coVC number} of a family $\H$, denoted by $\covcmod(\H)$, is the largest integer $d$ such that there exists a set $S\subset \X$, of size $d$ such that $\H_{|S}$ contains a hollow $d$-star, i.e. there is a subset $T\subset S$ such that \[ T\notin \H_{|S} \,\mbox{ and }\, \forall x\in S,\, T\Delta \set*{x} \in \H_{|S}\,. \]
      \end{definition}
      \begin{itemize}
      \item {\bf Bit-flipping invariant}: Y
      \item {\bf Monotonicity Type}: pM
      \end{itemize}
